cmake_minimum_required(VERSION 3.20)
project(GrannyHouseCpp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(RAYLIB_VERSION 5.5)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib
  URL https://github.com/raysan5/raylib/archive/refs/tags/${RAYLIB_VERSION}.tar.gz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(raylib)
add_executable(granny_house src/game8.cpp)
target_link_libraries(granny_house PRIVATE raylib)
if(WIN32)
  target_compile_definitions(granny_house PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
install(TARGETS granny_house RUNTIME DESTINATION . BUNDLE DESTINATION .)
